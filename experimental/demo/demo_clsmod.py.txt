class A:
    attr = 420
    def __new__(cls, *args, **kwargs):
        print(f'A.__new__({cls}, ...)')
        return super().__new__(cls, *args, **kwargs)
    def __init__(self, *args, **kwargs):
        print(f'A.__init__({self}, ...)')
        super().__init__(*args, **kwargs)
    def __del__(self):
        print(f'A.__del__({self})')
        # super().__del__()
    def __delattr__(self, name):
        print(f'A.__delattr__({self}, {name})')
        # super().__delattr__(name)
    def get(self, name):
        value = self.__getattribute__(name)
        print(f'{self}.__getattribute__{name} == {value}')
        return value
    def set(self, name, value):
        self.__setattr__(name, value)
        print(f'{self}.__setattr__({name}, {value})')

class B(A):
    def __init__(self):
        print(f'B.__init__({self})')
